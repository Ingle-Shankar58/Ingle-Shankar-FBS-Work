CREATE TABLE Patient (
    patient_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    date_of_birth DATE NOT NULL,
    gender ENUM('Male','Female','Other') NOT NULL,
    blood_group VARCHAR(50),
    contact_number VARCHAR(20),
    email VARCHAR(100),
    address TEXT,
    registration_date date
);

CREATE TABLE EMDICAL_HISTORY(
	HISTORY_ID INT PRIMARY KEY AUTO_INCREMENT,
   	patient_id INT,
	diagnosis VARCHAR(250),
	diagnosis_date DATE,
	past_treatment TEXT,
	Allergy varchar(100),
	record_date datetime,
	INSTRUCTIONS TEXT,
        CONSTRAINT SHANKY FOREIGN KEY(patient_id) REFERENCES PATIENT(PATIENT_ID)
	);

CREATE TABLE Patient_Insurance(
	Patient_Insurance_ID INT PRIMARY KEY AUTO_INCREMENT,
	patient_id INT,
	insurance_id INT ,
	FOREIGN KEY(patient_id) REFERENCES PATIENT(patient_id),
	FOREIGN KEY(insurance_id) REFERENCES Insurance(insurance_id)
	);
	
CREATE TABLE insurance (
	insurance_id  INT PRIMARY KEY,
	provider_name VARCHAR(250) NOT NULL,
	policy_number VARCHAR(100) NOT NULL,
	coverage_amount DECIMAL(10,2),
	issue_date DATE NOT NULL,
	expire_date DATE NOT NULL
	);

CREATE TABLE Doctor (
    doctor_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    gender ENUM('Male','Female','Other'),
    years_of_experience INT,
    joining_date DATE,
    specialization VARCHAR(250),
    contact_number VARCHAR(20),
    email VARCHAR(100)
);

CREATE TABLE Room (
    room_id INT PRIMARY KEY AUTO_INCREMENT,
    room_number VARCHAR(10),
    room_type ENUM('General','Semi-Private','Private','ICU'),
    status ENUM('Available','Occupied') DEFAULT 'Available'
);


CREATE TABLE Admission (
    admission_id INT PRIMARY KEY AUTO_INCREMENT,
    room_id INT,
    admission_date DATETIME,
    discharge_date DATETIME,
    FOREIGN KEY (room_id) REFERENCES Room(room_id)
);


CREATE TABLE Encounter (
    encounter_id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT,
    doctor_id INT,
    encounter_type ENUM('Appointment','Admission','Discharge'),
    encounter_date DATETIME,
    admission_id INT,
    notes TEXT,
    FOREIGN KEY (patient_id) REFERENCES Patient(patient_id),
    FOREIGN KEY (doctor_id) REFERENCES Doctor(doctor_id),
    FOREIGN KEY (admission_id) REFERENCES Admission(admission_id)
);
CREATE TABLE Medications (
    medication_id INT PRIMARY KEY AUTO_INCREMENT,
    dosage VARCHAR(50),
    frequency VARCHAR(50),
    duration VARCHAR(50),
    medication_name VARCHAR(100),
    instructions TEXT
);


CREATE TABLE Prescription (
    prescription_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT,
    prescribed_date DATE,
    medications_id INT,
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id),
    FOREIGN KEY (medications_id) REFERENCES Medications(medication_id)
);


CREATE TABLE Procedure_Surgery (
    procedure_surgery_id INT PRIMARY KEY AUTO_INCREMENT,
    surgery_name VARCHAR(100),
    surgery_type VARCHAR(100),
    surgery_date DATETIME,
    surgery_result VARCHAR(100)
);

CREATE TABLE Patient_Procedure (
    procedure_record_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT,
    procedure_id INT,
    procedure_date DATE,
    followup_required VARCHAR(100),
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id),
    FOREIGN KEY (procedure_id) REFERENCES Procedure_Surgery(procedure_surgery_id)
);

CREATE TABLE Patient_Procedure (
    procedure_record_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT,
    procedure_id INT,
    procedure_date DATE,
    followup_required VARCHAR(100),
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id),
    FOREIGN KEY (procedure_id) REFERENCES Procedure_Surgery(procedure_surgery_id)
);

CREATE TABLE Diagnostic_Test (
    diagnostic_test_id INT PRIMARY KEY AUTO_INCREMENT,
    test_name VARCHAR(100),
    diagnostic_test_date DATETIME,
    test_result_date DATE,
    diagnostic_test_result VARCHAR(100),
    description TEXT
);

CREATE TABLE Patient_Test (
    test_id INT PRIMARY KEY AUTO_INCREMENT,
    encounter_id INT,
    diagnostic_test_id INT,
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id),
    FOREIGN KEY (diagnostic_test_id) REFERENCES Diagnostic_Test(diagnostic_test_id)
);
